function drawGauge(spentValue, budgetValue, title) { document.getElementById('gaugeTitle').textContent = title; var canvas = document.getElementById('gauge'); var ctx = canvas.getContext('2d'); var centerX = canvas.width / 2; var centerY = canvas.height; var radius = 120; ctx.clearRect(0, 0, canvas.width, canvas.height); ctx.beginPath(); ctx.arc(centerX, centerY, radius, Math.PI, 0, false); ctx.lineWidth = 15; ctx.strokeStyle = "#e6e6e6"; ctx.stroke(); ctx.closePath(); var proportionSpent = spentValue / budgetValue; ctx.beginPath(); ctx.arc(centerX, centerY, radius, Math.PI, Math.PI + (Math.PI * proportionSpent), false); ctx.lineWidth = 15; ctx.strokeStyle = "#29b50d"; ctx.stroke(); ctx.closePath(); var angle = Math.PI + (Math.PI * proportionSpent); ctx.beginPath(); ctx.moveTo(centerX, centerY); ctx.lineTo(centerX + radius * Math.cos(angle), centerY + radius * Math.sin(angle)); ctx.lineWidth = 4; ctx.strokeStyle = "#333"; ctx.stroke(); ctx.closePath(); ctx.beginPath(); ctx.arc(centerX, centerY, 8, 0, 2 * Math.PI); ctx.fillStyle = "#333"; ctx.fill(); ctx.closePath(); ctx.font = "bold 20px Arial"; ctx.fillStyle = "#333"; ctx.fillText(spentValue.toLocaleString() + " / " + budgetValue.toLocaleString(), centerX, centerY - 30); } function listenForUpdates() { db.ref('gaugeData').on('value', (snapshot) => { const data = snapshot.val(); const spentValue = data ? data.spentValue : 0; const budgetValue = 7000000; drawGauge(spentValue, budgetValue, "Budget vs. Spent"); }); } listenForUpdates(); document.getElementById('updateForm').addEventListener('submit', function(e) { e.preventDefault(); const newSpentValue = parseFloat(document.getElementById('spentInput').value); if (!isNaN(newSpentValue)) { db.ref('gaugeData').set({ spentValue: newSpentValue }); } });
